# 号池管理(Account Pool)实施计划

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** 在 proapi 现有 channel 体系上增量构建账号池层,支持 Codex 类多账号批量入池、OAuth/Token-JSON/API Key 三类凭证、上游限额嗅探、token 自动刷新、账号级熔断与完整监控。

**Architecture:** 新增独立模块 `internal/account/` 与 `internal/channel/` 平级;数据层 = 主表 `accounts` + 事件表 `account_events` + `channels` 增 3 字段;运行时层 = Selector / Breaker / Refresher / Probe / QuotaTracker / Importer / OAuthFlow 七组件,通过 `WireAccount` 装配到 `app.Application`;relay 链路在 channel 选定后判断 `pool_enabled` 分支取 account。

**Tech Stack:** Go 1.22 + GORM + MySQL 8 / PostgreSQL 16 + Redis 7(SET NX 锁、SADD 池缓存) + AES-256-GCM(凭证加密) + Gin(REST) + asynq(可选,后台 worker 进程内 goroutine 即可) + Vue 3 + Vite + TS + Naive UI(admin 前端) + dockertest(集成测试)

**M1 范围:** Task 1 - Task 10 (P1-P10),Task 11 (P11 OAuth 直连)留 M2。

## 设计原则

- **TDD:每个 Task 都先写失败的测试,再写最小实现使其通过。**
- **DRY:重复出现 3 次的代码才抽公共函数;两次内联即可。**
- **YAGNI:M1 只实现 M1 范围的功能,M2 接口先留 stub,不要预想未来需求。**
- **频繁 commit:每个 Task 内部完成一个完整 vertical slice 就 commit;不要等整个 Task 完成再 commit。**
- **遵循现有代码风格:** Repo 沿用 `internal/channel/repo.go` 的 `repo{db, crypto, id, clock}` 结构;Facade/Wire 沿用 `internal/channel/wire.go` 的 `Facade + WireXxx(ctx, *app.Application)` 模式;handler 沿用 `internal/server/handler/admin/setting.go` 的 actorOf 模式。

```go
type repo struct {
    db     *gorm.DB
    crypto *crypto.AESGCM
    id     *idgen.Generator
    clock  clock.Clock
}

// NewRepository 构造 Repo。
func NewRepository(db *gorm.DB, c *crypto.AESGCM, id *idgen.Generator, clk clock.Clock) Repo {
    return &repo{db: db, crypto: c, id: id, clock: clk}
}

func (r *repo) Create(ctx context.Context, a *Account) error {
    if err := r.encryptCred(a); err != nil {
        return err
    }
    if a.ID == 0 {
        a.ID = r.id.Generate()
    }
    now := r.clock.Now().UTC()
    if a.CreatedAt.IsZero() {
        a.CreatedAt = now
    }
    a.UpdatedAt = now
    if len(a.Extra) == 0 {
        a.Extra = json.RawMessage("{}")
    }
    return r.db.WithContext(ctx).Create(a).Error
}

func (r *repo) Update(ctx context.Context, a *Account) error {
    if a.Cred != (AccountCred{}) {
        if err := r.encryptCred(a); err != nil {
            return err
        }
    }
    a.UpdatedAt = r.clock.Now().UTC()
    return r.db.WithContext(ctx).Save(a).Error
}

func (r *repo) Get(ctx context.Context, id int64) (*Account, error) {
    var a Account
    err := r.db.WithContext(ctx).Where("id = ? AND deleted_at IS NULL", id).First(&a).Error
    if err == gorm.ErrRecordNotFound {
        return nil, apierr.New(apierr.CodeAccountNotFound, "account not found")
    }
    if err != nil {
        return nil, err
    }
    if err := r.hydrate(&a); err != nil {
        return nil, err
    }
    return &a, nil
}

func (r *repo) ListByChannel(ctx context.Context, channelID int64) ([]*Account, error) {
    var out []*Account
    err := r.db.WithContext(ctx).
        Where("channel_id = ? AND deleted_at IS NULL", channelID).
        Order("priority DESC, weight DESC, id DESC").Find(&out).Error
    if err != nil {
        return nil, err
    }
    for _, a := range out {
        _ = r.hydrate(a)
    }
    return out, nil
}

func (r *repo) ListByShareTag(ctx context.Context, tag string) ([]*Account, error) {
    var out []*Account
    err := r.db.WithContext(ctx).
        Where("share_tag = ? AND deleted_at IS NULL", tag).
        Find(&out).Error
    if err != nil {
        return nil, err
    }
    for _, a := range out {
        _ = r.hydrate(a)
    }
    return out, nil
}

func (r *repo) ListForRefresher(ctx context.Context, before time.Time, limit int) ([]*Account, error) {
    var out []*Account
    err := r.db.WithContext(ctx).
        Where("cred_type = 'oauth' AND status = ? AND access_token_expires_at IS NOT NULL AND access_token_expires_at < ?",
            StatusActive, before).
        Limit(limit).Find(&out).Error
    if err != nil {
        return nil, err
    }
    for _, a := range out {
        _ = r.hydrate(a)
    }
    return out, nil
}

func (r *repo) ListForReaper(ctx context.Context, now time.Time, limit int) ([]*Account, error) {
    var out []*Account
    err := r.db.WithContext(ctx).
        Where("status = ? AND cooldown_until IS NOT NULL AND cooldown_until <= ?",
            StatusCooldown, now).
        Limit(limit).Find(&out).Error
    return out, err
}

func (r *repo) Delete(ctx context.Context, id int64) error {
    now := r.clock.Now().UTC()
    return r.db.WithContext(ctx).Model(&Account{}).
        Where("id = ? AND deleted_at IS NULL", id).
        Updates(map[string]any{"deleted_at": now, "updated_at": now}).Error
}

func (r *repo) AppendEvent(ctx context.Context, accountID int64, eventType string, payload any) error {
    var raw json.RawMessage
    if payload != nil {
        b, err := json.Marshal(payload)
        if err != nil {
            return apierr.New(apierr.CodeInvalidParam, "encode event payload: "+err.Error())
        }
        raw = b
    }
    ev := &AccountEvent{
        ID:        r.id.Generate(),
        AccountID: accountID,
        EventType: eventType,
        Payload:   raw,
        CreatedAt: r.clock.Now().UTC(),
    }
    return r.db.WithContext(ctx).Create(ev).Error
}

func (r *repo) encryptCred(a *Account) error {
    b, err := json.Marshal(a.Cred)
    if err != nil {
        return apierr.New(apierr.CodeInvalidParam, "encode cred: "+err.Error())
    }
    enc, err := r.crypto.Encrypt(string(b))
    if err != nil {
        return apierr.New(apierr.CodeInternal, "encrypt cred: "+err.Error())
    }
    a.Credentials = enc
    return nil
}

func (r *repo) hydrate(a *Account) error {
    if a.Credentials == "" {
        return nil
    }
    raw, err := r.crypto.Decrypt(a.Credentials)
    if err != nil {
        return apierr.New(apierr.CodeInternal,
            "decrypt cred failed for account "+strconv.FormatInt(a.ID, 10))
    }
    return json.Unmarshal([]byte(raw), &a.Cred)
}
```

```go
func (a *Anthropic) Extract(h http.Header) *account.QuotaSnapshot {
    snap := &account.QuotaSnapshot{}
    parse5h(h, &snap.Quota5h, "anthropic-ratelimit-tokens-limit",

    snap.QuotaWeek.SyncedAt = &now
    return snap
}

func parse5h(h http.Header, w *account.QuotaWindow, totalKey, remKey, resetKey string) {
    if v := h.Get(totalKey); v != "" {
        if n, err := strconv.ParseInt(v, 10, 64); err == nil {
            w.Total = &n
        }
    }
    if v := h.Get(remKey); v != "" {
        if n, err := strconv.ParseInt(v, 10, 64); err == nil {
            w.Remaining = &n
        }
    }
    if v := h.Get(resetKey); v != "" {
        if t, err := time.Parse(time.RFC3339, v); err == nil {
            w.ResetAt = &t
        }
    }
}
```

```go
var reDur = regexp.MustCompile(`(?i)^(\d+)(ms|s|m|h|d)$`)

func parseDuration(s string) (time.Duration, bool) {
    m := reDur.FindStringSubmatch(s)
    if m == nil {
        if d, err := time.ParseDuration(s); err == nil {
            return d, true
        }
        return 0, false
    }
    n, _ := strconv.ParseInt(m[1], 10, 64)
    switch m[2] {
    case "ms":
        return time.Duration(n) * time.Millisecond, true
    case "s":
        return time.Duration(n) * time.Second, true
    case "m":
        return time.Duration(n) * time.Minute, true
    case "h":
        return time.Duration(n) * time.Hour, true
    case "d":
        return time.Duration(n) * 24 * time.Hour, true
    }
    return 0, false
}
```

```go
package account

import (
    "context"
    "net/http"

// providerQuota 是各 provider 实现的接口(由 quota/ 子包提供)。
type providerQuota interface {
    Extract(h http.Header) *QuotaSnapshot
}

// quotaTracker 默认 QuotaTracker 实现,持有 provider 注册表。
type quotaTracker struct {
    repo      Repo
    providers map[string]providerQuota
}

// NewQuotaTracker 构造 QuotaTracker。providers 由 wire 注入。
func NewQuotaTracker(r Repo, providers map[string]providerQuota) QuotaTracker {
    return &quotaTracker{repo: r, providers: providers}
}

func (t *quotaTracker) ExtractFromResponse(provider string, h http.Header) *QuotaSnapshot {
    p, ok := t.providers[provider]
    if !ok {
        return nil
    }
```

```go
package account

import (
    "bytes"
    "context"
    "encoding/json"
    "errors"
    "strings"

// FormatParser 是各格式的解析器接口。
type FormatParser interface {
    Format() string
    Match(payload []byte) bool
    Parse(ctx context.Context, payload []byte) ([]*Account, error)
}

type importer struct {
    parsers []FormatParser
    oauth   OAuthFlow // 用于 refresh_token → access_token 换取
}

// NewImporter 接收 OAuthFlow(可为 nil,仅做格式探测)。
func NewImporter(oauth OAuthFlow) Importer {
    return &importer{oauth: oauth}
}

// RegisterParser 注入一个解析器(wire 时按顺序传入)。
func (i *importer) RegisterParser(p FormatParser) {
    i.parsers = append(i.parsers, p)
}

func (i *importer) Detect(payload []byte) (string, bool) {
    trimmed := bytes.TrimSpace(payload)
    for _, p := range i.parsers {
        if p.Match(trimmed) {
            return p.Format(), true
        }
    }
    return "", false
}

func (i *importer) Parse(payload []byte, format string) ([]*Account, error) {
    trimmed := bytes.TrimSpace(payload)
    for _, p := range i.parsers {
        if p.Format() == format {
            return p.Parse(context.Background(), trimmed)
        }
    }
    return nil, apierr.New(apierr.CodeAccountImportFormat, "unknown format: "+format)
}

// helper:严格 JSON 解析。
func decodeJSON(b []byte, dst any) error {
    dec := json.NewDecoder(bytes.NewReader(b))
    dec.DisallowUnknownFields()
    dec = json.NewDecoder(bytes.NewReader(b)) // 二次构造,允许未知字段(默认)
    return dec.Decode(dst)
}

// isRawString:不是 JSON 开头(`{` `[`)的纯字符串。
func isRawString(b []byte) bool {
    if len(b) == 0 {
        return false
    }
    s := strings.TrimSpace(string(b))
    return !strings.HasPrefix(s, "{") && !strings.HasPrefix(s, "[")
}
```

```go
    providers    map[string]ProviderProbe

func NewProbe(r Repo, q QuotaTracker, providers map[string]ProviderProbe) Probe {

func (p *probeImpl) Run(ctx context.Context, a *Account) error {

    if snap != nil {

    }
    _ = p.repo.AppendEvent(ctx, a.ID, "probed", snap)
```

```go
type selectorImpl struct {
    repo    Repo
    breaker Breaker
    rnd     *rand.Rand
    mu      sync.Mutex
}

func NewSelector(r Repo, b Breaker, seed int64) Selector {
    return &selectorImpl{repo: r, breaker: b, rnd: rand.New(rand.NewSource(seed))}
}

func (s *selectorImpl) Select(ctx context.Context, ch *channel.Channel, hint SelectHint) (*Account, error) {
    pool, err := s.poolFor(ctx, ch)
    if err != nil {
        return nil, err
    }
    cands := make([]*Account, 0, len(pool))
    for _, a := range pool {
        if a.Status != StatusActive {
            continue
        }
        if containsInt64(hint.Excluded, a.ID) {
            continue
        }
        cands = append(cands, a)
    }
    if len(cands) == 0 {
        return nil, ErrPoolEmpty
    }

    strategy := ch.AccountStrategy
    if strategy == "" {
        strategy = "top_k"
    }
    switch strategy {
    case "top_k":
        return s.topK(cands, int(ch.AccountTopK)), nil
    case "round_robin":
        return s.roundRobin(cands, ch.ID), nil
    case "weighted":
        return s.weighted(cands), nil
    case "max_remaining":
        return s.maxRemaining(cands), nil
    default:
        return s.topK(cands, int(ch.AccountTopK)), nil
    }
}

func (s *selectorImpl) topK(cands []*Account, k int) *Account {
    sort.SliceStable(cands, func(i, j int) bool {
        return remPct(cands[i]) > remPct(cands[j])
    })
    if k <= 0 || k > len(cands) {
        k = len(cands)
    }
    return s.weighted(cands[:k])
}

func (s *selectorImpl) maxRemaining(cands []*Account) *Account {
    best := cands[0]
    for _, c := range cands[1:] {
        if remPct(c) > remPct(best) {
            best = c
        }
    }
    return best
}

func (s *selectorImpl) weighted(cands []*Account) *Account {
    total := 0
    for _, c := range cands {
        if c.Weight <= 0 {
            c.Weight = 1
        }
        total += c.Weight
    }
    s.mu.Lock()
    n := s.rnd.Intn(total)
    s.mu.Unlock()
    acc := 0
    for _, c := range cands {
        acc += c.Weight
        if n < acc {
            return c
        }
    }
    return cands[len(cands)-1]
}

var rrCursor sync.Map // channelID -> int

func (s *selectorImpl) roundRobin(cands []*Account, channelID int64) *Account {
    sort.Slice(cands, func(i, j int) bool { return cands[i].ID < cands[j].ID })
    cur, _ := rrCursor.LoadOrStore(channelID, 0)
    n := cur.(int) % len(cands)
    rrCursor.Store(channelID, n+1)
    return cands[n]
}

func (s *selectorImpl) poolFor(ctx context.Context, ch *channel.Channel) ([]*Account, error) {
    own, err := s.repo.ListByChannel(ctx, ch.ID)
    if err != nil {
        return nil, err
    }
    tags := []string{}
    if len(ch.Tags) > 0 {
        _ = json.Unmarshal(ch.Tags, &tags)
    }
    seen := map[int64]struct{}{}
    out := []*Account{}
    for _, a := range own {
        seen[a.ID] = struct{}{}
        out = append(out, a)
    }
    for _, t := range tags {
        if !strings.HasPrefix(t, "share:") {
            continue
        }
        shared, err := s.repo.ListByShareTag(ctx, strings.TrimPrefix(t, "share:"))
        if err != nil {
            continue
        }
        for _, a := range shared {
            if _, ok := seen[a.ID]; ok {
                continue
            }
            seen[a.ID] = struct{}{}
            out = append(out, a)
        }
    }
    return out, nil
}

func remPct(a *Account) float64 {
    if a.Quota5hTotal == nil || a.Quota5hRemaining == nil || *a.Quota5hTotal == 0 {
        return 1.0
    }
    return float64(*a.Quota5hRemaining) / float64(*a.Quota5hTotal)
}

func containsInt64(xs []int64, x int64) bool {
    for _, v := range xs {
        if v == x {
            return true
        }
    }
    return false
}

func (s *selectorImpl) ReportSuccess(accountID int64, latency time.Duration) {
    now := time.Now().UTC()
    _ = s.repo.Update(context.Background(), &Account{
        ID:            accountID,
        LastSuccessAt: &now,
        LastUsedAt:    &now,
        ConsecFailures: 0,
    })
}

func (s *selectorImpl) ReportFailure(accountID int64, err error, headers http.Header) {
    if s.breaker == nil {
        return
    }
    ctx := context.Background()
    code := classifyFailure(err)
    switch code {
    case failRateLimited:
        reset := parseResetAt(headers)
        if reset.IsZero() {
            reset = time.Now().Add(5 * time.Minute)
        }
        _ = s.breaker.MarkCooldown(ctx, accountID, reset, err.Error())
    case failTokenExpired:
        _ = s.breaker.MarkExpired(ctx, accountID, err.Error())
    case failInvalidCred:
        _ = s.breaker.MarkInvalid(ctx, accountID, err.Error())
    default:
        n, _ := s.breaker.IncConsecFailure(ctx, accountID)
        if n >= 5 {
            _ = s.breaker.MarkCooldown(ctx, accountID, time.Now().Add(60*time.Second), "consec fail >=5")
        }
    }
}
```

```go
type failClass int

const (
    failUnknown failClass = iota
    failRateLimited
    failTokenExpired
    failInvalidCred
)

func classifyFailure(err error) failClass {
    if err == nil {
        return failUnknown
    }
    s := strings.ToLower(err.Error())
    switch {
    case strings.Contains(s, "429") || strings.Contains(s, "rate limit"):
        return failRateLimited
    case strings.Contains(s, "401") || strings.Contains(s, "token expired") || strings.Contains(s, "invalid_token"):
        return failTokenExpired
    case strings.Contains(s, "403") || strings.Contains(s, "invalid_grant") || strings.Contains(s, "revoked"):
        return failInvalidCred
    }
    return failUnknown
}

func parseResetAt(h http.Header) time.Time {
    for _, k := range []string{"anthropic-ratelimit-tokens-reset", "x-ratelimit-reset-tokens", "retry-after"} {
        if v := h.Get(k); v != "" {
            if t, err := time.Parse(time.RFC3339, v); err == nil {
                return t
            }
            if d, ok := parseSeconds(v); ok {
                return time.Now().Add(d)
            }
        }
    }
    return time.Time{}
}

func parseSeconds(s string) (time.Duration, bool) {
    var n int
    _, err := fmt.Sscanf(s, "%d", &n)
    if err != nil {
        return 0, false
    }
    return time.Duration(n) * time.Second, true
}
```

```go
import (
```

```go
type breakerImpl struct {
    repo  Repo
    cache *redis.Client
    log   *zap.Logger
    stop  chan struct{}
}

func NewBreaker(r Repo, c *redis.Client, log *zap.Logger) Breaker {
    return &breakerImpl{repo: r, cache: c, log: log, stop: make(chan struct{})}
}

func (b *breakerImpl) MarkCooldown(ctx context.Context, id int64, until time.Time, reason string) error {
    a, err := b.repo.Get(ctx, id)
    if err != nil {
        return err
    }
    a.Status = StatusCooldown
    a.CooldownUntil = &until
    a.LastFailureReason = trunc(reason, 256)
    now := time.Now().UTC()
    a.LastFailureAt = &now
    if err := b.repo.Update(ctx, a); err != nil {
        return err
    }
    return b.repo.AppendEvent(ctx, id, "cooldown_entered", map[string]any{"until": until, "reason": reason})
}

func (b *breakerImpl) MarkExpired(ctx context.Context, id int64, reason string) error {
    return b.setStatus(ctx, id, StatusExpired, reason, "refresh_failed")
}

func (b *breakerImpl) MarkInvalid(ctx context.Context, id int64, reason string) error {
    return b.setStatus(ctx, id, StatusInvalid, reason, "marked_invalid")
}

func (b *breakerImpl) setStatus(ctx context.Context, id int64, st Status, reason, evType string) error {
    a, err := b.repo.Get(ctx, id)
    if err != nil {
        return err
    }
    a.Status = st
    a.LastFailureReason = trunc(reason, 256)
    now := time.Now().UTC()
    a.LastFailureAt = &now
    if err := b.repo.Update(ctx, a); err != nil {
        return err
    }
    return b.repo.AppendEvent(ctx, id, evType, map[string]any{"reason": reason})
}

func (b *breakerImpl) IncConsecFailure(ctx context.Context, id int64) (int, error) {
    a, err := b.repo.Get(ctx, id)
    if err != nil {
        return 0, err
    }
    a.ConsecFailures++
    if err := b.repo.Update(ctx, a); err != nil {
        return 0, err
    }
    return a.ConsecFailures, nil
}

func (b *breakerImpl) Run(ctx context.Context) error {
    tick := time.NewTicker(30 * time.Second)
    defer tick.Stop()
    for {
        select {
        case <-ctx.Done():
            return nil
        case <-b.stop:
            return nil
        case <-tick.C:
            n, err := b.RunReaperOnce(ctx)
            if err != nil && b.log != nil {
                b.log.Warn("breaker reaper", zap.Error(err))
            }
            if n > 0 && b.log != nil {
                b.log.Info("breaker reaped", zap.Int("count", n))
            }
        }
    }
}

func (b *breakerImpl) RunReaperOnce(ctx context.Context) (int, error) {
    now := time.Now().UTC()
    list, err := b.repo.ListForReaper(ctx, now, 100)
    if err != nil {
        return 0, err
    }
    for _, a := range list {
        a.Status = StatusActive
        a.CooldownUntil = nil
        a.ConsecFailures = 0
        _ = b.repo.Update(ctx, a)
        _ = b.repo.AppendEvent(ctx, a.ID, "cooldown_exited", nil)
    }
    return len(list), nil
}

func (b *breakerImpl) Close() error {
    close(b.stop)
    return nil
}

func trunc(s string, n int) string {
    if len(s) <= n {
        return s
    }
    return s[:n]
}
```

```go
package account

import (
    "context"
    "errors"
    "fmt"
    "time"

    "github.com/redis/go-redis/v9"
    "go.uber.org/zap"
)

type refresherImpl struct {
    repo  Repo
    cache *redis.Client
    oauth OAuthFlow
    log   *zap.Logger
    stop  chan struct{}
}

func NewRefresher(r Repo, c *redis.Client, o OAuthFlow, log *zap.Logger) Refresher {
    return &refresherImpl{repo: r, cache: c, oauth: o, log: log, stop: make(chan struct{})}
}

func (r *refresherImpl) Run(ctx context.Context) error {
    tick := time.NewTicker(60 * time.Second)
    defer tick.Stop()
    for {
        select {
        case <-ctx.Done():
            return nil
        case <-r.stop:
            return nil
        case <-tick.C:
            before := time.Now().Add(5 * time.Minute)
            list, err := r.repo.ListForRefresher(ctx, before, 100)
            if err != nil {
                if r.log != nil {
                    r.log.Warn("refresher list", zap.Error(err))
                }
                continue
            }
            for _, a := range list {
                go func(id int64) { _ = r.RefreshOne(ctx, id) }(a.ID)
            }
        }
    }
}

func (r *refresherImpl) RefreshOne(ctx context.Context, accountID int64) error {
    lockKey := fmt.Sprintf("account:lock:refresh:%d", accountID)
    if r.cache != nil {
        ok, err := r.cache.SetNX(ctx, lockKey, "1", 30*time.Second).Result()
        if err != nil {
            return err
        }
        if !ok {
            return errors.New("refresh lock held")
        }
        defer r.cache.Del(ctx, lockKey)
    }

    a, err := r.repo.Get(ctx, accountID)
    if err != nil {
        return err
    }
    if a.CredType != "oauth" || a.Cred.RefreshToken == "" {
        return nil
    }
    newCred, err := r.oauth.ExchangeRefreshToken(ctx, a.Provider, a.Cred.RefreshToken)
    if err != nil {
        a.RefreshTokenValid = 2
        _ = r.repo.Update(ctx, a)
        _ = r.repo.AppendEvent(ctx, accountID, "refresh_failed", map[string]any{"err": err.Error()})
        return err
    }
    a.Cred = *newCred
    a.RefreshTokenValid = 1
    now := time.Now().UTC()
    a.LastRefreshedAt = &now
    if !newCred.ExpiresAt.IsZero() {
        t := newCred.ExpiresAt
        a.AccessTokenExpiresAt = &t
    }
    if err := r.repo.Update(ctx, a); err != nil {
        return err
    }
    return r.repo.AppendEvent(ctx, accountID, "token_refreshed", nil)
}
```

```go
type Service struct {
    registry        adapter.Registry
    accountFacade   AccountFacade // 可为 nil(channel 未启用池)
}

// AccountFacade 是 account 模块向 relay 暴露的最小接口,避免循环 import。
type AccountFacade interface {
    Select(ctx context.Context, ch *channel.Channel, hint account.SelectHint) (*account.Account, error)
    ReportSuccess(accountID int64, latency time.Duration)
    ReportFailure(accountID int64, err error, headers http.Header)
}

func New(reg adapter.Registry, af AccountFacade) *Service {
    return &Service{registry: reg, accountFacade: af}
}

func (s *Service) Chat(ctx context.Context, req *ir.ChatRequest, ch *channel.Channel) (*ir.ChatResponse, int64, error) {
    cred := ch.Cred.ToAdapter()
    var accID int64
    if ch.PoolEnabled == 1 && s.accountFacade != nil {
        a, err := s.accountFacade.Select(ctx, ch, account.SelectHint{Model: req.Model})
        if err != nil {
            return nil, 0, err
        }
        cred = adapter.Credential{APIKey: a.Cred.AccessToken, BaseURL: ch.BaseURL}
        if a.Cred.APIKey != "" {
            cred.APIKey = a.Cred.APIKey
        }
        accID = a.ID
    }
    ad, ok := s.registry.Get(ch.Provider)
    if !ok {
        return nil, 0, fmt.Errorf("relay: unknown provider %q", ch.Provider)
    }
    start := time.Now()
    resp, err := ad.Chat(ctx, req, cred)
    if accID > 0 && s.accountFacade != nil {
        if err != nil {
            var h http.Header
            if resp != nil {
                h = resp.Headers
            }
            s.accountFacade.ReportFailure(accID, err, h)
        } else {
            s.accountFacade.ReportSuccess(accID, time.Since(start))
        }
    }
    return resp, accID, err
}
```

```go
package account

import (
    "context"
    "errors"
    "time"

    "github.com/ijry/pro-api/internal/account/importer"
    "github.com/ijry/pro-api/internal/account/oauth"

    "github.com/ijry/pro-api/internal/account/quota"
    "github.com/ijry/pro-api/internal/app"
    "github.com/ijry/pro-api/internal/util/clock"
)

// FacadeFrom 提取已 wire 的 Facade(供其它模块、handler 用)。
func FacadeFrom(a *app.Application) *Facade {
    if f, ok := a.AccountSvc.(*Facade); ok {
        return f
    }
    return nil
}

// WireAccount 装配账号池。
func WireAccount(ctx context.Context, a *app.Application) error {
    if a.DB == nil {
        return errors.New("account: DB nil")
    }
    if a.Cache == nil {
        return errors.New("account: Cache nil")
    }
    clk := a.Clock
    if clk == nil {
        clk = clock.Real
    }
    repo := NewRepository(a.DB, a.Crypto, a.IDGen, clk)

    oa := oauth.NewFlow(map[string]oauth.Provider{
        "anthropic": oauth.NewAnthropic(oauth.Config{
            TokenURL: a.Config.OAuthAnthropicTokenURL,
            ClientID: a.Config.OAuthAnthropicClientID,
        }),
        "openai": oauth.NewOpenAI(oauth.Config{
            TokenURL: a.Config.OAuthOpenAITokenURL,
            ClientID: a.Config.OAuthOpenAIClientID,
        }),
    })
    tracker := quota.NewTracker(repo)
    pr := NewProbe(repo, tracker, map[string]ProviderProbe{
        "anthropic": probe.NewAnthropic(a.Config.AnthropicProbeBase),
        "openai":    probe.NewOpenAI(a.Config.OpenAIProbeBase),
    })
    br := NewBreaker(repo, a.Cache, a.Log)
    go func() { _ = br.Run(ctx) }()

    sel := NewSelector(repo, br, time.Now().UnixNano())
    rf := NewRefresher(repo, a.Cache, oa, a.Log)
    go func() { _ = rf.Run(ctx) }()

    imp := newImporterImpl(oa)
    imp.RegisterParser(importer.OAuthSession{})
    imp.RegisterParser(importer.ClaudeAuthJSON{})
    imp.RegisterParser(importer.CodexAuthJSON{})
    imp.RegisterParser(importer.Sub2API{})
    imp.RegisterParser(importer.CLIProxy{})
    imp.RegisterParser(importer.AuthsBatch{})
    imp.RegisterParser(importer.RawAccessToken{})
    imp.RegisterParser(importer.RawRefreshToken{OAuth: oa})
    imp.RegisterParser(importer.RawAPIKey{})

    a.AccountSvc = &Facade{
        Repo: repo, Selector: sel, Breaker: br,
        Refresher: rf, Probe: pr, Importer: imp,
        OAuth: oa, QuotaTrack: tracker,
    }
    a.AddCloser("account_breaker", br.Close)
    return nil
}

// newImporterImpl 构造可注册 parser 的 importer。
func newImporterImpl(oa OAuthFlow) *importerImpl {
    return &importerImpl{oauth: oa}
}
```

```go
type createReq struct {
    ChannelID int64                  `json:"channel_id" binding:"required"`
    Name      string                 `json:"name"`
    Provider  string                 `json:"provider"`
    Format    string                 `json:"format"`
    Text      string                 `json:"text"`
    DryRun    bool                   `json:"dry_run"`
    ShareTag  string                 `json:"share_tag"`

func (h *AccountHandler) Create(c *gin.Context) {
    var req createReq
    if err := c.ShouldBindJSON(&req); err != nil { respondErr(c, apierr.New(apierr.CodeInvalidParam, err.Error())); return }
    format := req.Format
    if format == "" {
        f, ok := h.facade.Importer.Detect([]byte(req.Text))
        if !ok { respondErr(c, apierr.New(apierr.CodeAccountImportFormat, "cannot detect format")); return }
        format = f
    }
    list, err := h.facade.Importer.Parse([]byte(req.Text), format)
    if err != nil { respondErr(c, err); return }
    if len(list) == 0 { respondErr(c, apierr.New(apierr.CodeAccountImportFields, "empty parse result")); return }
    a := list[0]
    a.ChannelID = req.ChannelID
    a.Name = req.Name
    a.ShareTag = req.ShareTag
    if req.DryRun {
        respondOK(c, gin.H{"preview": accountToListItem(a)}); return
    }
    if err := h.facade.Repo.Create(c.Request.Context(), a); err != nil { respondErr(c, err); return }
    _ = h.facade.Repo.AppendEvent(c.Request.Context(), a.ID, "imported", map[string]any{"by": h.actorOf(c), "format": format})
    go func() { _ = h.facade.Probe.Run(c.Copy().Request.Context(), a) }()
    h.audit.Write(c.Request.Context(), audit.Entry{Actor: h.actorOf(c), Action: "account.create", Target: a.ID})
    respondOK(c, gin.H{"id": a.ID})
}

// POST /api/admin/accounts/import (multipart 或 json)
func (h *AccountHandler) Import(c *gin.Context) {
    // 与 Create 类似,但循环处理 N 条 + dedup + 收集 errors。
    // 实现略,逻辑见 spec §4.2。
    var req createReq
    if err := c.ShouldBindJSON(&req); err != nil { respondErr(c, apierr.New(apierr.CodeInvalidParam, err.Error())); return }
    text := req.Text
    if file, _ := c.FormFile("file"); file != nil {
        f, _ := file.Open()
        defer f.Close()
        buf := make([]byte, file.Size)
        _, _ = f.Read(buf)
        text = string(buf)
    }
    format := req.Format
    if format == "" {
        format, _ = h.facade.Importer.Detect([]byte(text))
    }
    list, err := h.facade.Importer.Parse([]byte(text), format)
    if err != nil { respondErr(c, err); return }

    var imported []int64
    var skipped int
    var errs []gin.H
    for i, a := range list {
        a.ChannelID = req.ChannelID
        a.ShareTag = req.ShareTag
        if req.DryRun {
            continue
        }
        if err := h.facade.Repo.Create(c.Request.Context(), a); err != nil {
            errs = append(errs, gin.H{"index": i, "reason": err.Error()})
            continue
        }
        imported = append(imported, a.ID)
    }
    respondOK(c, gin.H{
        "total": len(list), "imported": len(imported),
        "skipped": skipped, "errors": errs, "account_ids": imported,
    })
}
```

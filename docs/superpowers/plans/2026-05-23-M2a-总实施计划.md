```go
// Package anthropic 处理 Anthropic Messages API 的入口协议编解码。
//
// 注意：这是入口解码（客户端发过来的 Anthropic 格式 → IR）。
// 与 internal/adapter/anthropic（后端适配器，把 IR 发给 Claude API）是两个不同的包。
package anthropic

import (
	"encoding/json"
	"fmt"

	"github.com/ijry/pro-api/internal/protocol/ir"
)

// anthropicRequest 是 Anthropic Messages API 请求结构。
type anthropicRequest struct {
	Model     string           `json:"model"`
	MaxTokens int              `json:"max_tokens"`
	System    string           `json:"system,omitempty"`
	Messages  []anthropicMsg   `json:"messages"`
	Stream    bool             `json:"stream,omitempty"`
	Tools     []anthropicTool  `json:"tools,omitempty"`
	ToolChoice any              `json:"tool_choice,omitempty"`
	Temperature *float64       `json:"temperature,omitempty"`
	TopP        *float64       `json:"top_p,omitempty"`
	TopK        *int           `json:"top_k,omitempty"`
	StopSequences []string     `json:"stop_sequences,omitempty"`
	Metadata    map[string]any `json:"metadata,omitempty"`
}

type anthropicMsg struct {
	Role    string          `json:"role"`
	Content json.RawMessage `json:"content"` // string or []block
}

type anthropicBlock struct {
	Type   string         `json:"type"`
	Text   string         `json:"text,omitempty"`
	Source *anthropicSrc  `json:"source,omitempty"`
	// tool_use fields
	ID    string         `json:"id,omitempty"`
	Name  string         `json:"name,omitempty"`
	Input map[string]any `json:"input,omitempty"`
	// tool_result fields
	ToolUseID string          `json:"tool_use_id,omitempty"`
	Content   json.RawMessage `json:"content,omitempty"`
}

type anthropicSrc struct {
	Type      string `json:"type"` // "url" | "base64"
	URL       string `json:"url,omitempty"`
	MediaType string `json:"media_type,omitempty"`
	Data      string `json:"data,omitempty"`
}

type anthropicTool struct {
	Name        string         `json:"name"`
	Description string         `json:"description,omitempty"`
	InputSchema map[string]any `json:"input_schema"`
}

// DecodeRequest 把 Anthropic Messages API 请求体解析为 IR。
func DecodeRequest(body []byte) (*ir.ChatRequest, error) {
	var ar anthropicRequest
	if err := json.Unmarshal(body, &ar); err != nil {
		return nil, fmt.Errorf("anthropic decode: %w", err)
	}

	req := &ir.ChatRequest{
		Model:     ar.Model,
		MaxTokens: ar.MaxTokens,
		Stream:    ar.Stream,
	}
	if ar.Temperature != nil {
		req.Temperature = ar.Temperature
	}
	if ar.TopP != nil {
		req.TopP = ar.TopP
	}
	if len(ar.StopSequences) > 0 {
		req.Stop = ar.StopSequences
	}

	// system → 第一条 system message
	if ar.System != "" {
		req.Messages = append(req.Messages, ir.Message{
			Role:    "system",
			Content: []ir.ContentPart{{Type: "text", Text: ar.System}},
		})
	}

	// 转换 messages
	for _, m := range ar.Messages {
		irMsg, err := convertMsg(m)
		if err != nil {
			return nil, err
		}
		req.Messages = append(req.Messages, irMsg)
	}

	// 转换 tools
	for _, t := range ar.Tools {
		req.Tools = append(req.Tools, ir.Tool{
			Type: "function",
			Function: ir.FunctionTool{
				Name:        t.Name,
				Description: t.Description,
				Parameters:  t.InputSchema,
			},
		})
	}
	if ar.ToolChoice != nil {
		req.ToolChoice = ar.ToolChoice
	}

	return req, nil
}

func convertMsg(m anthropicMsg) (ir.Message, error) {
	irMsg := ir.Message{Role: m.Role}

	// content 可能是 string 或 []block
	var strContent string
	if err := json.Unmarshal(m.Content, &strContent); err == nil {
		irMsg.Content = []ir.ContentPart{{Type: "text", Text: strContent}}
		return irMsg, nil
	}

	var blocks []anthropicBlock
	if err := json.Unmarshal(m.Content, &blocks); err != nil {
		return irMsg, fmt.Errorf("anthropic decode: parse content: %w", err)
	}

	for _, b := range blocks {
		switch b.Type {
		case "text":
			irMsg.Content = append(irMsg.Content, ir.ContentPart{Type: "text", Text: b.Text})
		case "image":
			part := ir.ContentPart{Type: "image_url"}
			if b.Source != nil {
				if b.Source.Type == "url" {
					part.ImageURL = ir.ImageURL{URL: b.Source.URL}
				} else {
					// base64
					part.ImageURL = ir.ImageURL{
						URL: "data:" + b.Source.MediaType + ";base64," + b.Source.Data,
					}
				}
			}
			irMsg.Content = append(irMsg.Content, part)
		case "tool_use":
			inputJSON, _ := json.Marshal(b.Input)
			irMsg.ToolCalls = append(irMsg.ToolCalls, ir.ToolCall{
				ID:   b.ID,
				Type: "function",
				Function: ir.ToolCallFunction{
					Name:      b.Name,
					Arguments: string(inputJSON),
				},
			})
		case "tool_result":
			// tool_result → role=tool message
			// 内容可以是 string 或 []block
			var resultText string
			if err := json.Unmarshal(b.Content, &resultText); err == nil {
				irMsg.Content = append(irMsg.Content, ir.ContentPart{Type: "text", Text: resultText})
			}
			irMsg.ToolCallID = b.ToolUseID
			irMsg.Role = "tool"
		}
	}
	return irMsg, nil
}
```

```go
package anthropic

import (
	"encoding/json"
	"fmt"

	"github.com/ijry/pro-api/internal/protocol/ir"
)

// SSEEvent 是一条 Anthropic SSE 事件。
type SSEEvent struct {
	Event string
	Data  []byte
}

// EncodeResponse 把 IR 响应转为 Anthropic Messages API 响应 JSON。
func EncodeResponse(resp *ir.ChatResponse) ([]byte, error) {
	if len(resp.Choices) == 0 {
		return nil, fmt.Errorf("anthropic encode: no choices")
	}
	choice := resp.Choices[0]

	type textBlock struct {
		Type string `json:"type"`
		Text string `json:"text"`
	}
	type toolUseBlock struct {
		Type  string         `json:"type"`
		ID    string         `json:"id"`
		Name  string         `json:"name"`
		Input map[string]any `json:"input"`
	}

	var content []any
	for _, part := range choice.Message.Content {
		if part.Type == "text" && part.Text != "" {
			content = append(content, textBlock{Type: "text", Text: part.Text})
		}
	}
	for _, tc := range choice.Message.ToolCalls {
		var input map[string]any
		_ = json.Unmarshal([]byte(tc.Function.Arguments), &input)
		content = append(content, toolUseBlock{
			Type:  "tool_use",
			ID:    tc.ID,
			Name:  tc.Function.Name,
			Input: input,
		})
	}
	if len(content) == 0 {
		content = []any{textBlock{Type: "text", Text: ""}}
	}

	stopReason := mapStopReason(choice.FinishReason)

	out := map[string]any{
		"id":      resp.ID,
		"type":    "message",
		"role":    "assistant",
		"model":   resp.Model,
		"content": content,
		"stop_reason":    stopReason,
		"stop_sequence":  nil,
		"usage": map[string]int{
			"input_tokens":  resp.Usage.PromptTokens,
			"output_tokens": resp.Usage.CompletionTokens,
		},
	}
	return json.Marshal(out)
}

// EncodeChunk 把 IR chunk 转为 Anthropic SSE events。
// index 是 content_block 的索引，isStop 标记是否是最后一个 chunk。
func EncodeChunk(chunk *ir.ChatChunk, index int, isStop bool) ([]SSEEvent, error) {
	var events []SSEEvent

	if chunk.Delta.Content != "" {
		data, _ := json.Marshal(map[string]any{
			"type":  "content_block_delta",
			"index": index,
			"delta": map[string]string{"type": "text_delta", "text": chunk.Delta.Content},
		})
		events = append(events, SSEEvent{Event: "content_block_delta", Data: data})
	}

	if isStop || chunk.FinishReason != "" {
		stopReason := mapStopReason(chunk.FinishReason)
		if stopReason == "" {
			stopReason = "end_turn"
		}

		deltaData, _ := json.Marshal(map[string]any{
			"type":        "message_delta",
			"delta":       map[string]string{"type": "message_delta", "stop_reason": stopReason},
			"usage":       buildUsage(chunk.Usage),
		})
		events = append(events, SSEEvent{Event: "message_delta", Data: deltaData})

		stopData, _ := json.Marshal(map[string]string{"type": "message_stop"})
		events = append(events, SSEEvent{Event: "message_stop", Data: stopData})
	}

	return events, nil
}

func mapStopReason(r string) string {
	switch r {
	case "stop", "":
		return "end_turn"
	case "length":
		return "max_tokens"
	case "tool_calls", "tool_use":
		return "tool_use"
	default:
		return r
	}
}

func buildUsage(u *ir.Usage) map[string]int {
	if u == nil {
		return map[string]int{"output_tokens": 0}
	}
	return map[string]int{
		"input_tokens":  u.PromptTokens,
		"output_tokens": u.CompletionTokens,
	}
}
```

```go
// Package gemini 处理 Gemini API 的入口协议编解码。
package gemini

import (
	"encoding/json"
	"fmt"

	"github.com/ijry/pro-api/internal/protocol/ir"
)

type geminiRequest struct {
	Contents            []geminiContent    `json:"contents"`
	SystemInstruction   *geminiContent     `json:"system_instruction,omitempty"`
	Tools               []geminiTool       `json:"tools,omitempty"`
	GenerationConfig    *generationConfig  `json:"generationConfig,omitempty"`
	SafetySettings      []any              `json:"safetySettings,omitempty"`
	Stream              bool               `json:"-"` // 由 URL 路径决定
}

type geminiContent struct {
	Role  string       `json:"role,omitempty"`
	Parts []geminiPart `json:"parts"`
}

type geminiPart struct {
	Text         string             `json:"text,omitempty"`
	InlineData   *geminiInlineData  `json:"inline_data,omitempty"`
	FunctionCall *geminiFuncCall    `json:"function_call,omitempty"`
	FunctionResponse *geminiFuncResp `json:"function_response,omitempty"`
}

type geminiInlineData struct {
	MIMEType string `json:"mime_type"`
	Data     string `json:"data"`
}

type geminiFuncCall struct {
	Name string         `json:"name"`
	Args map[string]any `json:"args"`
}

type geminiFuncResp struct {
	Name     string         `json:"name"`
	Response map[string]any `json:"response"`
}

type geminiTool struct {
	FunctionDeclarations []geminiFuncDecl `json:"function_declarations"`
}

type geminiFuncDecl struct {
	Name        string         `json:"name"`
	Description string         `json:"description"`
	Parameters  map[string]any `json:"parameters"`
}

type generationConfig struct {
	MaxOutputTokens int     `json:"maxOutputTokens,omitempty"`
	Temperature     float64 `json:"temperature,omitempty"`
	TopP            float64 `json:"topP,omitempty"`
	TopK            int     `json:"topK,omitempty"`
	StopSequences   []string `json:"stopSequences,omitempty"`
}

// DecodeRequest 把 Gemini generateContent 请求体解析为 IR。
// model 由 URL 路径参数传入。
func DecodeRequest(body []byte, model string) (*ir.ChatRequest, error) {
	var gr geminiRequest
	if err := json.Unmarshal(body, &gr); err != nil {

			Role:    "system",
			Content: []ir.ContentPart{{Type: "text", Text: gr.SystemInstruction.Parts[0].Text}},
		})
	}

	// contents
	for _, c := range gr.Contents {
		role := c.Role
		if role == "model" {
			role = "assistant"
		}
		msg := ir.Message{Role: role}
		for _, p := range c.Parts {
			if p.Text != "" {
				msg.Content = append(msg.Content, ir.ContentPart{Type: "text", Text: p.Text})
			}
			if p.InlineData != nil {

					Type: "function",
					Function: ir.ToolCallFunction{
						Name:      p.FunctionCall.Name,
						Arguments: string(argsJSON),
```

```go
// EncodeResponse 把 IR 响应转为 Gemini generateContent 响应 JSON。
func EncodeResponse(resp *ir.ChatResponse) ([]byte, error) {
	if len(resp.Choices) == 0 {
		return nil, fmt.Errorf("gemini encode: no choices")
	}
	choice := resp.Choices[0]

	var parts []map[string]any
	for _, p := range choice.Message.Content {
		if p.Type == "text" {
			parts = append(parts, map[string]any{"text": p.Text})
		}
	}
	for _, tc := range choice.Message.ToolCalls {
		var args map[string]any
		_ = json.Unmarshal([]byte(tc.Function.Arguments), &args)
		parts = append(parts, map[string]any{
			"functionCall": map[string]any{"name": tc.Function.Name, "args": args},
		})
	}
	if len(parts) == 0 {
		parts = []map[string]any{{"text": ""}}
	}

	finishReason := mapFinishReason(choice.FinishReason)

	out := map[string]any{
		"candidates": []map[string]any{
			{
				"content":      map[string]any{"role": "model", "parts": parts},
				"finishReason": finishReason,
				"index":        0,
			},
		},
		"usageMetadata": map[string]int{
			"promptTokenCount":     resp.Usage.PromptTokens,
			"candidatesTokenCount": resp.Usage.CompletionTokens,
			"totalTokenCount":      resp.Usage.TotalTokens,
		},
	}
	return json.Marshal(out)
}

// SSEEvent Gemini SSE 流事件。
type SSEEvent struct {
	Data []byte
}

// EncodeChunk 把 IR chunk 转为 Gemini SSE 流事件。
func EncodeChunk(chunk *ir.ChatChunk, isStop bool) ([]SSEEvent, error) {
	var parts []map[string]any
	if chunk.Delta.Content != "" {
		parts = append(parts, map[string]any{"text": chunk.Delta.Content})
	}
	finishReason := ""
	if isStop || chunk.FinishReason != "" {
		finishReason = mapFinishReason(chunk.FinishReason)
		if finishReason == "" {
			finishReason = "STOP"
		}
	}

	candidate := map[string]any{
		"content": map[string]any{"role": "model", "parts": parts},
		"index":   0,
	}
	if finishReason != "" {
		candidate["finishReason"] = finishReason
	}
	out := map[string]any{"candidates": []any{candidate}}
	if isStop && chunk.Usage != nil {
		out["usageMetadata"] = map[string]int{
			"promptTokenCount":     chunk.Usage.PromptTokens,
			"candidatesTokenCount": chunk.Usage.CompletionTokens,
		}
	}
	data, err := json.Marshal(out)
	if err != nil {
		return nil, err
	}
	return []SSEEvent{{Data: data}}, nil
}

func mapFinishReason(r string) string {
	switch r {
	case "stop", "":
		return "STOP"
	case "length":
		return "MAX_TOKENS"
	case "tool_calls":
		return "STOP" // Gemini 不区分
	default:
		return "STOP"
	}
}
```

```go
		var idx int

			}
			if nextErr != nil {
				break
			}
			isStop := chunk.FinishReason != ""
			events, _ := anthropicprot.EncodeChunk(chunk, idx, isStop)
			for _, ev := range events {
				fmt.Fprintf(w, "event: %s\ndata: %s\n\n", ev.Event, ev.Data)
			}
			if flusher != nil {
				flusher.Flush()

			}

	}

		return
```

```go
func (a *Adapter) buildRequest(req *ir.ChatRequest) cohereChatReq {
	cr := cohereChatReq{
		Model:     req.Model,
		MaxTokens: req.MaxTokens,
		Stream:    false,
	}
	if req.Temperature != nil {
		cr.Temperature = *req.Temperature
	}

		role := m.Role
		if role == "assistant" {
			role = "assistant"
		}
		var blocks []cohereBlock
		for _, p := range m.Content {
			if p.Type == "text" {
				blocks = append(blocks, cohereBlock{Type: "text", Text: p.Text})
			}
		}
		cr.Messages = append(cr.Messages, cohereMsg{Role: role, Content: blocks})
	}
	return cr
}

func (a *Adapter) toIRResponse(cr *cohereChatResp, model string) *ir.ChatResponse {
	var content []ir.ContentPart
	for _, b := range cr.Message.Content {
		if b.Type == "text" {
			content = append(content, ir.ContentPart{Type: "text", Text: b.Text})
		}
	}
	resp := &ir.ChatResponse{
		ID:    cr.ID,
		Model: model,
		Choices: []ir.Choice{{
			Message:      ir.Message{Role: "assistant", Content: content},
			FinishReason: mapCoherefinish(cr.FinishReason),
		}},
	}
	if t, ok := cr.Usage.Tokens["input_tokens"]; ok {
		resp.Usage.PromptTokens = t
	}
	if t, ok := cr.Usage.Tokens["output_tokens"]; ok {
		resp.Usage.CompletionTokens = t
	}
	resp.Usage.TotalTokens = resp.Usage.PromptTokens + resp.Usage.CompletionTokens
	return resp
}

	switch r {
	case "COMPLETE":

	case "MAX_TOKENS":
		return "length"
	case "TOOL_CALL":
		return "tool_calls"
	default:
		return "stop"
	}
```

```go
// NewSingleChunkReader 把一个完整响应包装成只有一个 chunk 的 StreamReader，用于不支持流式的 adapter。
func NewSingleChunkReader(resp *ir.ChatResponse) StreamReader {
    if len(resp.Choices) == 0 {
        return &singleChunkReader{}
    }
    ch := &ir.ChatChunk{
        ID:    resp.ID,
        Model: resp.Model,
        Delta: ir.Delta{},
    }
    if len(resp.Choices) > 0 {
        ch.FinishReason = resp.Choices[0].FinishReason
        if len(resp.Choices[0].Message.Content) > 0 {
            ch.Delta.Content = resp.Choices[0].Message.Content[0].Text
        }
    }
    ch.Usage = &resp.Usage
    return &singleChunkReader{chunk: ch, done: false}
}

type singleChunkReader struct {
    chunk *ir.ChatChunk
    done  bool
}

func (r *singleChunkReader) Next(ctx context.Context) (*ir.ChatChunk, error) {
    if r.done || r.chunk == nil {
        return nil, io.EOF
    }
    r.done = true
    return r.chunk, nil
}

func (r *singleChunkReader) Close() error { return nil }
```

```go
import (
```

```go
// SMTPMailer 是生产环境 SMTP 邮件发送器。
// 配置来自 system_settings，支持 TLS/STARTTLS。
type SMTPMailer struct {
	Host     string
	Port     int
	Username string
	Password string
	From     string
	TLS      bool
	Log      *zap.Logger
}

// Send 发送邮件。实现 Mailer 接口。
func (m *SMTPMailer) Send(ctx context.Context, msg Message) error {
	if m.Host == "" {
		return fmt.Errorf("smtp: host not configured")
	}

	// 构建邮件内容
	header := make(map[string]string)
	header["From"] = mail.Address{Name: "proapi", Address: m.From}.String()
	header["To"] = strings.Join(msg.To, ", ")
	header["Subject"] = msg.Subject
	header["MIME-Version"] = "1.0"
	header["Content-Type"] = "text/html; charset=UTF-8"
	header["Date"] = time.Now().Format(time.RFC1123Z)

	var body strings.Builder
	for k, v := range header {
		body.WriteString(k)
		body.WriteString(": ")
		body.WriteString(v)
		body.WriteString("\r\n")
	}
	body.WriteString("\r\n")
	if msg.HTML != "" {
		body.WriteString(msg.HTML)
	} else {
		body.WriteString(msg.Text)
	}

	addr := fmt.Sprintf("%s:%d", m.Host, m.Port)
	auth := smtp.PlainAuth("", m.Username, m.Password, m.Host)

	if m.TLS {
		return m.sendTLS(addr, auth, msg.To, body.String())
	}
	return smtp.SendMail(addr, auth, m.From, msg.To, []byte(body.String()))
}

func (m *SMTPMailer) sendTLS(addr string, auth smtp.Auth, to []string, body string) error {
	host, _, _ := net.SplitHostPort(addr)
	tlsCfg := &tls.Config{ServerName: host}
	conn, err := tls.Dial("tcp", addr, tlsCfg)
	if err != nil {
		return fmt.Errorf("smtp tls dial: %w", err)
	}
	defer conn.Close()

	client, err := smtp.NewClient(conn, host)
	if err != nil {
		return fmt.Errorf("smtp client: %w", err)
	}
	defer client.Close()

	if err := client.Auth(auth); err != nil {
		return fmt.Errorf("smtp auth: %w", err)
	}
	if err := client.Mail(m.From); err != nil {
		return fmt.Errorf("smtp mail from: %w", err)
	}
	for _, addr := range to {
		if err := client.Rcpt(addr); err != nil {
			return fmt.Errorf("smtp rcpt: %w", err)
		}
	}
	w, err := client.Data()
	if err != nil {
		return fmt.Errorf("smtp data: %w", err)
	}
	defer w.Close()
	_, err = w.Write([]byte(body))
	return err
}
```

```go
func (a *Adapter) GenerateImage(ctx context.Context, req *ir.ImageRequest, cred Credential) (*ir.ImageResponse, error) {
	body, _ := json.Marshal(map[string]any{

// Transcribe 调用 Whisper STT。
func (a *Adapter) Transcribe(ctx context.Context, req *ir.TranscribeRequest, cred Credential) (*ir.TranscribeResponse, error) {
	// 使用 multipart/form-data
	var b bytes.Buffer
	w := multipart.NewWriter(&b)
	fw, _ := w.CreateFormFile("file", req.Filename)
	fw.Write(req.AudioData)
	w.WriteField("model", req.Model)
	if req.Language != "" { w.WriteField("language", req.Language) }
	if req.Format != "" { w.WriteField("response_format", req.Format) }
	w.Close()

	httpReq, _ := http.NewRequestWithContext(ctx, http.MethodPost, a.baseURL+"/audio/transcriptions", &b)
	httpReq.Header.Set("Authorization", "Bearer "+cred.APIKey)
	httpReq.Header.Set("Content-Type", w.FormDataContentType())

	resp, err := a.httpClient.Do(httpReq)
	if err != nil {
		return nil, apierr.Wrap(apierr.CodeUpstreamError, "openai transcribe", err)
	}
	defer resp.Body.Close()

	var out struct {
		Text string `json:"text"`
	}
	json.NewDecoder(resp.Body).Decode(&out)
	return &ir.TranscribeResponse{Text: out.Text}, nil
}
```

```go
// ImageGenerations 处理 POST /v1/images/generations (DALL-E)
func (h *Handler) ImageGenerations(c *gin.Context) {
	body, err := io.ReadAll(io.LimitReader(c.Request.Body, 1*1024*1024))
	if err != nil {
		middleware.SetErr(c, apierr.New(apierr.CodeInvalidParam, "read body: "+err.Error()))
		return
	}

	var req struct {
		Model   string `json:"model"`
		Prompt  string `json:"prompt"`
		N       int    `json:"n"`
		Size    string `json:"size"`
		Quality string `json:"quality"`
		Style   string `json:"style"`
		Format  string `json:"response_format"`
	}
	if err := json.Unmarshal(body, &req); err != nil {
		middleware.SetErr(c, apierr.New(apierr.CodeInvalidParam, "parse body: "+err.Error()))
		return
	}
	if req.N == 0 { req.N = 1 }
	if req.Format == "" { req.Format = "url" }

	irReq := &ir.ImageRequest{
		Model:  req.Model,
		Prompt: req.Prompt,
		N:      req.N,
		Size:   req.Size,
		Quality: req.Quality,
		Style:  req.Style,
		Format: req.Format,
	}

	// 找能支持图像生成的 adapter
	adp, ok := h.registry.Get(defaultProvider(irReq.Model))
	if !ok {
		middleware.SetErr(c, apierr.New(apierr.CodeNoChannel, "no adapter for "+irReq.Model))
		return
	}
	ia, ok := adp.(adapter.ImageAdapter)
	if !ok {
		middleware.SetErr(c, apierr.New(apierr.CodeModelNotSupported, "adapter does not support image generation"))
		return
	}

	resp, err := ia.GenerateImage(c.Request.Context(), irReq, adapter.Credential{})
	if err != nil {
		middleware.SetErr(c, toApierr(err))
		return
	}

	// 转换为 OpenAI 响应格式
	data := make([]map[string]any, 0, len(resp.Data))
	for _, d := range resp.Data {
		item := map[string]any{}
		if d.URL != "" { item["url"] = d.URL }
		if d.B64JSON != "" { item["b64_json"] = d.B64JSON }
		if d.RevisedPrompt != "" { item["revised_prompt"] = d.RevisedPrompt }
		data = append(data, item)
	}
	c.JSON(http.StatusOK, gin.H{"created": resp.Created, "data": data})
}

// AudioTranscriptions 处理 POST /v1/audio/transcriptions (STT)
func (h *Handler) AudioTranscriptions(c *gin.Context) {
	if err := c.Request.ParseMultipartForm(25 << 20); err != nil {
		middleware.SetErr(c, apierr.New(apierr.CodeInvalidParam, "parse form: "+err.Error()))
		return
	}

	file, header, err := c.Request.FormFile("file")
	if err != nil {
		middleware.SetErr(c, apierr.New(apierr.CodeMissingParam, "missing file"))
		return
	}
	defer file.Close()
	audio, _ := io.ReadAll(file)

	model := c.Request.FormValue("model")
	if model == "" { model = "whisper-1" }

	adp, ok := h.registry.Get(defaultProvider(model))
	if !ok {
		middleware.SetErr(c, apierr.New(apierr.CodeNoChannel, "no adapter for "+model))
		return
	}
	aa, ok := adp.(adapter.AudioAdapter)
	if !ok {
		middleware.SetErr(c, apierr.New(apierr.CodeModelNotSupported, "adapter does not support STT"))
		return
	}

	resp, err := aa.Transcribe(c.Request.Context(), &ir.TranscribeRequest{
		Model:     model,
		AudioData: audio,
		Filename:  header.Filename,
		Language:  c.Request.FormValue("language"),
		Format:    c.Request.FormValue("response_format"),
	}, adapter.Credential{})
	if err != nil {
		middleware.SetErr(c, toApierr(err))
		return
	}

}
```
